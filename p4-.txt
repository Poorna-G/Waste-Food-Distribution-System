 4. Create Views

Open:

core/views.py



from django.shortcuts import render, redirect
from .models import FoodPost
from .forms import FoodPostForm

# Home page - list all food posts
def home(request):
    foods = FoodPost.objects.all()
    return render(request, 'core/home.html', {'foods': foods})


# Add food post
def add_food(request):
    if request.method == 'POST':
        form = FoodPostForm(request.POST)
        if form.is_valid():
            food = form.save(commit=False)
            food.donor = request.user   # logged-in user
            food.save()
            return redirect('home')
    else:
        form = FoodPostForm()

    return render(request, 'core/add_food.html', {'form': form})

5.core/models.py
from django.db import models
from django.contrib.auth.models import User

class UserProfile(models.Model):
    ROLE_CHOICES = (
        ('donor', 'Donor'),
        ('volunteer', 'Volunteer'),
    )

    user = models.OneToOneField(User, on_delete=models.CASCADE)
    role = models.CharField(max_length=20, choices=ROLE_CHOICES)

    def __str__(self):
        return self.user.username


class FoodPost(models.Model):
    donor = models.ForeignKey(User, on_delete=models.CASCADE)
    food_name = models.CharField(max_length=100)
    quantity = models.CharField(max_length=50)
    location = models.CharField(max_length=200)
    expiry_time = models.DateTimeField()
    description = models.TextField(blank=True)
    status = models.CharField(max_length=20, default='Available')

    def __str__(self):
        return self.food_name


class Request(models.Model):
    food_post = models.ForeignKey(FoodPost, on_delete=models.CASCADE)
    volunteer = models.ForeignKey(User, on_delete=models.CASCADE)
    status = models.CharField(max_length=20, default='Requested')


class Pickup(models.Model):
    request = models.OneToOneField(Request, on_delete=models.CASCADE)
    pickup_time = models.DateTimeField()
    delivery_status = models.CharField(max_length=20, default='Pending')

10.update core/views.py
from django.shortcuts import render, redirect
from .models import FoodPost
from .forms import FoodPostForm
from django.contrib.auth.models import User


# ✅ HOME VIEW (IMPORTANT - was missing)
def home(request):
    foods = FoodPost.objects.all()
    return render(request, 'core/home.html', {'foods': foods})


# ✅ ADD FOOD VIEW
def add_food(request):
    if request.method == 'POST':
        form = FoodPostForm(request.POST)
        if form.is_valid():
            food = form.save(commit=False)

            # 🔥 SAFETY CHECK
            user = User.objects.first()

            if user is None:
                return render(request, 'core/add_food.html', {
                    'form': form,
                    'error': 'No user found. Please create a user first.'
                })

            food.donor = user
            food.save()

            return redirect('home')
    else:
        form = FoodPostForm()

    return render(request, 'core/add_food.html', {'form': form})
